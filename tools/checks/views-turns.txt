# The views' turns on the connection while one of them is held up: run with --url text.html --viewport 400x300@1 on a
# server with --expose-internals-object.
settle 0.5
{"type":"create_view","width":200,"height":150,"dpr":1,"url":"$pages/animate.html","request":2}
expect view_created view=2
expect-frame view=2 full=true
sleep 0.5

# A page that sets a title of 1 MiB over and over while the client reads nothing neither costs the client its
# connection nor keeps the other views' frames behind its titles.
{"type":"load_html","view":1,"html":"<script>let i = 0; const big = 'x'.repeat(1 << 20); const timer = setInterval(() => { document.title = (i++) + big; if (i == 500) { clearInterval(timer); document.title = 'done'; } }, 0)</script>"}
stall 6
sleep 0.5
frames 0 100000 view=2
sleep 1
frames 5 10000 view=2
expect state view=1 title=done timeout=60
{"type":"ping","id":"alive"}
expect pong id=alive

# A view whose frames the client does not acknowledge holds up no other view.
ack off view=1
{"type":"load","view":1,"url":"$pages/animate.html"}
expect state view=1 title=animate loading=false
sleep 0.5
frames 0 10000 view=1
frames 0 10000 view=2
sleep 1
frames 0 1 view=1
frames 5 10000 view=2
ack on view=1
sleep 0.5
frames 3 10000 view=1

# While a band of one view's frame is not acknowledged, the other views wait for the rest of that frame. Hiding the
# view ends its frame, and the others go on; shown again, it starts with a whole frame.
{"type":"load","view":1,"url":"$pages/text.html"}
expect state view=1 title=text loading=false
sleep 1
ack off view=1 bands
{"type":"set_viewport","view":1,"width":4200,"height":4100,"dpr":1}
expect-frame view=1 full=true more=true width=4200 height=4100
frames 0 10000 view=2
expect-no-frame 1 view=2
{"type":"set_visibility","view":1,"visible":false}
expect-frame view=2
sleep 0.5
frames 5 10000 view=2
ack on view=1
{"type":"set_visibility","view":1,"visible":true}
expect-frame view=1 full=true more=true width=4200 height=4100
expect-frame view=1 more=false width=4200 height=4100
sleep 0.5
