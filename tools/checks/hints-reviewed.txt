# Pages a review used to break hints, each loaded in turn: run with --url hints-shift.html --viewport 800x600@1.

# A page that moves the labels' host puts the labels on other elements, so nothing is activated.
{"type":"set_focus","focused":true}
settle 0.5
{"type":"hints","id":1,"alphabet":"asdf"}
expect hints_list id=1 items#=2
settle 0.7
color-in 400 200 400 400 fff3a3 6
hint "one"
{"type":"hints_end","id":1,"activate":"$hint.index"}
expect error id=1 "message~=tampered"
expect-no state 1
