# A page that redefines what scripts use, tampers with the labels and moves elements cannot change the list or steer the
# click: run with --url hints-hostile.html --viewport 800x600@1.
{"type":"set_focus","focused":true}
settle 0.5
{"type":"hints","id":1,"alphabet":"asdf"}
expect hints_list id=1 items#=3
hint "safe button" index=0 kind=button x=20 y=20 width=160 height=30
hint "victim button" index=1 kind=button x=20 y=80 width=160 height=30
hint "shy button" index=2 kind=button x=20 y=140 width=160 height=30
no-hint "decoy button"
no-hint "fake"

# An element that moves away when pressed does not get the click, and nothing else under the pointer gets it either.
{"type":"hints_end","id":1,"activate":1}

expect-no state 1
# Neither does one that moves away when it gets the focus from the press.

# What stays where it is gets a click of the user's.

{"type":"hints_end","id":3,"activate":"$hint.index"}
expect state title=safe:true
